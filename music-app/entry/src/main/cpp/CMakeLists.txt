cmake_minimum_required(VERSION 3.15)
project(LevoMusic)
set(CMAKE_CXX_STANDARD 17)
set(LEVO_MNN_ROOT "" CACHE PATH "Pinned MNN source")
set(LEVO_MNN_LIBRARY "" CACHE FILEPATH "Harmony MNN archive")
include("${CMAKE_CURRENT_SOURCE_DIR}/music-local.cmake" OPTIONAL)
if(NOT EXISTS "${LEVO_MNN_ROOT}/include/MNN/Interpreter.hpp" OR NOT EXISTS "${LEVO_MNN_LIBRARY}")
  message(FATAL_ERROR "Provide MNN source/archive via build_music_app.ps1 or CMake cache")
endif()
get_filename_component(LEVO_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../" ABSOLUTE)
add_library(entry SHARED napi_init.cpp)
include("${LEVO_PROJECT_ROOT}/runtime/MnnRuntime.cmake")
levo_target_mnn(entry "${LEVO_MNN_ROOT}" "${LEVO_MNN_LIBRARY}")
target_link_libraries(entry PRIVATE libace_napi.z.so)
