cmake_minimum_required(VERSION 3.15)
project(LevoBlockBench CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MNN_ROOT "" CACHE PATH "Pinned MNN source tree")
set(MNN_STATIC_LIBRARY "" CACHE FILEPATH "Compatible ARM64 Harmony MNN archive")
if(NOT EXISTS "${MNN_ROOT}/include/MNN/Interpreter.hpp" OR NOT EXISTS "${MNN_STATIC_LIBRARY}")
  message(FATAL_ERROR "Provide MNN_ROOT and MNN_STATIC_LIBRARY")
endif()
add_executable(levo_block_bench BlockBench.cpp)
add_executable(levo_vae_bench VaeBench.cpp)
add_executable(levo_vae_profile VaeProfile.cpp)
add_executable(levo_vae_kernel VaeBench.cpp)
add_executable(levo_vae_npu_probe VaeNpuProbe.cpp)
add_executable(levo_vae_snake_fusion VaeSnakeFusion.cpp)
target_include_directories(levo_vae_snake_fusion PRIVATE "${MNN_ROOT}/include" "${MNN_ROOT}/source" "${MNN_ROOT}/schema/current" "${MNN_ROOT}/3rd_party/flatbuffers/include" "${MNN_ROOT}/transformers/llm/engine/app/jsonhpp")
target_compile_options(levo_vae_snake_fusion PRIVATE -O3 -march=armv8.2-a+fp16 -ffp-contract=off)
target_compile_definitions(levo_vae_snake_fusion PRIVATE MNN_USE_THREAD_POOL)
target_link_libraries(levo_vae_snake_fusion PRIVATE "${MNN_STATIC_LIBRARY}" hilog_ndk.z)
target_include_directories(levo_vae_npu_probe PRIVATE "${MNN_ROOT}/include" "${MNN_ROOT}/transformers/llm/engine/app/jsonhpp")
target_compile_options(levo_vae_npu_probe PRIVATE -O3 -march=armv8.2-a+fp16)
target_link_libraries(levo_vae_npu_probe PRIVATE "${MNN_STATIC_LIBRARY}" hilog_ndk.z dl)
target_include_directories(levo_vae_profile PRIVATE "${MNN_ROOT}/schema/current" "${MNN_ROOT}/3rd_party/flatbuffers/include")
target_include_directories(levo_vae_kernel PRIVATE "${MNN_ROOT}/include" "${MNN_ROOT}/transformers/llm/engine/app/jsonhpp")
target_compile_options(levo_vae_kernel PRIVATE -O3 -fno-omit-frame-pointer)
target_link_options(levo_vae_kernel PRIVATE -Wl,--gc-sections)
target_link_libraries(levo_vae_kernel PRIVATE "${MNN_STATIC_LIBRARY}" hilog_ndk.z)
target_compile_definitions(levo_vae_kernel PRIVATE LEVO_VAE_VECTOR_SIN)
target_compile_options(levo_vae_kernel PRIVATE -march=armv8.2-a+fp16)
target_include_directories(levo_vae_kernel PRIVATE "${MNN_ROOT}/source" "${MNN_ROOT}/schema/current" "${MNN_ROOT}/3rd_party/flatbuffers/include")
add_executable(levo_chain_bench ChainBench.cpp)
add_executable(levo_chain_vae_optimized ChainBench.cpp)
target_include_directories(levo_chain_vae_optimized PRIVATE "${MNN_ROOT}/include" "${MNN_ROOT}/transformers/llm/engine/app/jsonhpp" "${MNN_ROOT}/transformers/llm/engine/include" "${MNN_ROOT}/source" "${MNN_ROOT}/schema/current" "${MNN_ROOT}/3rd_party/flatbuffers/include")
target_compile_definitions(levo_chain_vae_optimized PRIVATE LEVO_VAE_VECTOR_SIN)
target_compile_options(levo_chain_vae_optimized PRIVATE -O3 -march=armv8.2-a+fp16)
target_link_options(levo_chain_vae_optimized PRIVATE -Wl,--gc-sections -Wl,-s)
target_link_libraries(levo_chain_vae_optimized PRIVATE "${MNN_STATIC_LIBRARY}" hilog_ndk.z)
add_executable(levo_dit_profile DitProfile.cpp)
add_executable(levo_dit_low_profile DitProfile.cpp)
target_compile_definitions(levo_dit_low_profile PRIVATE LEVO_DIT_LOW_PROBE)
target_include_directories(levo_chain_bench PRIVATE "${MNN_ROOT}/transformers/llm/engine/include")
foreach(target levo_block_bench levo_vae_bench levo_vae_profile levo_chain_bench levo_dit_profile levo_dit_low_profile)
  target_include_directories(${target} PRIVATE "${MNN_ROOT}/include" "${MNN_ROOT}/transformers/llm/engine/app/jsonhpp")
  target_compile_options(${target} PRIVATE -O3)
  target_link_options(${target} PRIVATE -Wl,--gc-sections -Wl,-s)
  target_link_libraries(${target} PRIVATE "${MNN_STATIC_LIBRARY}" hilog_ndk.z)
endforeach()
