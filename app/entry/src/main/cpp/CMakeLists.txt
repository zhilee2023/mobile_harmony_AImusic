cmake_minimum_required(VERSION 3.5)
project(LevoHarmony)
set(CMAKE_CXX_STANDARD 17)
if(DEFINED PACKAGE_FIND_FILE)
  include(${PACKAGE_FIND_FILE})
endif()
add_library(entry SHARED napi_init.cpp)
target_compile_options(entry PRIVATE -O3)
target_link_libraries(entry PUBLIC libace_napi.z.so ohhidebug)
set(LEVO_MNN_ROOT "" CACHE PATH "Path to pinned MNN 3.6.1 source; empty builds diagnostics only")
set(LEVO_MNN_STATIC_LIBRARY "" CACHE FILEPATH "Existing compatible Harmony ARM64 MNN library")
include("${CMAKE_CURRENT_SOURCE_DIR}/levo-local.cmake" OPTIONAL)
if(LEVO_MNN_ROOT)
  if(LEVO_MNN_STATIC_LIBRARY)
    add_library(MNN STATIC IMPORTED)
    set_target_properties(MNN PROPERTIES IMPORTED_LOCATION "${LEVO_MNN_STATIC_LIBRARY}")
  else()
  set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
  set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
  set(MNN_LOW_MEMORY ON CACHE BOOL "" FORCE)
  set(MNN_ARM82 ON CACHE BOOL "" FORCE)
  set(MNN_OPENCL OFF CACHE BOOL "" FORCE)
  set(MNN_VULKAN OFF CACHE BOOL "" FORCE)
  add_subdirectory("${LEVO_MNN_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/mnn")
  endif()
  target_include_directories(entry PRIVATE "${LEVO_MNN_ROOT}/include")
  target_include_directories(entry PRIVATE "${LEVO_MNN_ROOT}/transformers/llm/engine/include")
  target_include_directories(entry PRIVATE "${LEVO_MNN_ROOT}/transformers/llm/engine/app/jsonhpp" "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native-bench")
  target_sources(entry PRIVATE mnn_compile_probe.cpp)
  target_sources(entry PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native-bench/VaeBench.cpp")
  target_sources(entry PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native-bench/ChainBench.cpp")
  target_compile_definitions(entry PRIVATE LEVO_VAE_LIBRARY LEVO_CHAIN_LIBRARY LEVO_WITH_MNN LEVO_VAE_VECTOR_SIN MNN_USE_THREAD_POOL MNN_SUPPORT_QUANT_EXTEND)
  target_compile_options(entry PRIVATE -march=armv8.2-a+fp16)
  target_include_directories(entry PRIVATE "${LEVO_MNN_ROOT}/source" "${LEVO_MNN_ROOT}/schema/current" "${LEVO_MNN_ROOT}/3rd_party/flatbuffers/include")
  target_link_libraries(entry PUBLIC MNN hilog_ndk.z)
endif()
