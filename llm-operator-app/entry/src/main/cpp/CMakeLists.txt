cmake_minimum_required(VERSION 3.15)
project(LlmOperatorLab)
set(CMAKE_CXX_STANDARD 17)
include("${CMAKE_CURRENT_SOURCE_DIR}/operator-local.cmake" OPTIONAL)
get_filename_component(LEVO_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../" ABSOLUTE)
add_library(entry SHARED napi_init.cpp)
include("${LEVO_PROJECT_ROOT}/runtime/MnnRuntime.cmake")
levo_target_mnn(entry "${LEVO_MNN_ROOT}" "${LEVO_MNN_LIBRARY}")
target_link_libraries(entry PRIVATE libace_napi.z.so)
file(SHA256 "${LEVO_MNN_LIBRARY}" OPERATOR_MNN_SHA256)
target_compile_definitions(entry PRIVATE LLM_OPERATOR_MNN_SHA256="${OPERATOR_MNN_SHA256}")
set(OPERATOR_CANDIDATE_SOURCE "" CACHE FILEPATH "Optional audited candidate source")
set(OPERATOR_VARIANT_ID "baseline" CACHE STRING "Candidate identity")
target_compile_definitions(entry PRIVATE LLM_OPERATOR_VARIANT_ID="${OPERATOR_VARIANT_ID}")
if(OPERATOR_CANDIDATE_SOURCE)
  if(NOT EXISTS "${OPERATOR_CANDIDATE_SOURCE}")
    message(FATAL_ERROR "Missing requested candidate source")
  endif()
  target_sources(entry PRIVATE "${OPERATOR_CANDIDATE_SOURCE}")
endif()
